#!/bin/bash set -euo pipefail section() { printf '\n=== %s ===\n' "$1"; } step() { printf '[%s] %s\n' "$1" "$2"; } err() { printf 'Error: %s\n' "$1" >&2; } PREFIX="${PREFIX:-/data/data/com.termux/files/usr}" USER_BIN="$HOME/.local/bin" USER_LIB="$HOME/.local/lib" CLAUDE_PKG_DIR="$USER_LIB/claude-code" CLAUDE_CLI="$CLAUDE_PKG_DIR/node_modules/@anthropic-ai/claude-code/cli.js" CLAUDE_VERSION_MARKER="$CLAUDE_PKG_DIR/.installed-version" CODEX_DIR="$USER_LIB/codex" CODEX_BIN="$CODEX_DIR/codex" CODEX_VERSION_MARKER="$CODEX_DIR/.installed-version" CODEX_RESOLV_CONF="$HOME/.config/codex/resolv.conf" AGY_DIR="$USER_LIB/antigravity" AGY_BIN="$AGY_DIR/agy" AGY_VA39_BIN="$AGY_DIR/agy.va39" AGY_VERSION_MARKER="$AGY_DIR/.installed-version" AGY_REPO="${AGY_REPO:-wallentx/antigravity-cli-termux}" CURL=(curl -fsSL --retry 3 --retry-delay 2) if [ $# -eq 0 ]; then printf '\nAction\n 1. Install\n 2. Uninstall\n\n' read -rp "Choice [1]: " raw_action raw_action="${raw_action:-1}" printf '\nTarget\n 1. All three\n 2. Claude Code\n 3. Codex CLI\n 4. Antigravity CLI\n\n' read -rp "Choice [1]: " raw_target raw_target="${raw_target:-1}" else raw_action="$1" raw_target="${2:-both}" fi case "$raw_action" in 1|install) action=install ;; 2|uninstall) action=uninstall ;; *) err "Invalid action: $raw_action"; exit 1 ;; esac case "$raw_target" in 1|all|both) DO_CLAUDE=1; DO_CODEX=1; DO_AGY=1 ;; 2|claude) DO_CLAUDE=1; DO_CODEX=0; DO_AGY=0 ;; 3|codex) DO_CLAUDE=0; DO_CODEX=1; DO_AGY=0 ;; 4|agy|antigravity) DO_CLAUDE=0; DO_CODEX=0; DO_AGY=1 ;; *) err "Invalid target: $raw_target"; exit 1 ;; esac TMP_DIR=$(mktemp -d) trap 'rm -rf "$TMP_DIR"' EXIT if [ "$action" = "install" ]; then step "0/1" "Checking prerequisites..." MISSING=() command -v curl >/dev/null 2>&1 || MISSING+=(curl) command -v rg >/dev/null 2>&1 || MISSING+=(ripgrep) command -v which >/dev/null 2>&1 || MISSING+=(which) command -v python3 >/dev/null 2>&1 || MISSING+=(python) command -v node >/dev/null 2>&1 || MISSING+=(nodejs) command -v objcopy >/dev/null 2>&1 || MISSING+=(binutils) command -v tar >/dev/null 2>&1 || MISSING+=(tar) [ -f "$PREFIX/etc/tls/cert.pem" ] || MISSING+=(ca-certificates) if [ ${#MISSING[@]} -gt 0 ]; then echo " Installing: ${MISSING[*]}" pkg install -y "${MISSING[@]}" >/dev/null 2>&1 || { err "failed to install prerequisites. Run manually:" echo " pkg install ${MISSING[*]}" >&2 exit 1 } else echo " All prerequisites present." fi if [ "$DO_AGY" = "1" ]; then step "0/2" "Checking Antigravity/Termux runtime..." if [ "$(uname -m)" != "aarch64" ]; then err "Antigravity Termux port currently requires aarch64 (found: $(uname -m))" exit 1 fi if [ ! -x "$PREFIX/glibc/lib/ld-linux-aarch64.so.1" ]; then echo " Installing Termux glibc repository/runtime..." pkg install -y glibc-repo >/dev/null 2>&1 || { err "Could not enable the Termux glibc repository (pkg install glibc-repo)" exit 1 } pkg install -y glibc >/dev/null 2>&1 || { err "Could not install Termux glibc runtime (pkg install glibc)" exit 1 } fi if [ ! -r "$PREFIX/etc/resolv.conf" ]; then echo " Installing Termux resolv-conf..." pkg install -y resolv-conf >/dev/null 2>&1 || { err "Could not install resolv-conf (pkg install resolv-conf)" exit 1 } fi if ! grep -q "atomics" /proc/cpuinfo 2>/dev/null && ! command -v qemu-aarch64 >/dev/null 2>&1; then echo " CPU does not advertise LSE atomics; installing qemu-user-aarch64 fallback..." pkg install -y qemu-user-aarch64 >/dev/null 2>&1 || { err "This CPU needs qemu-aarch64 for Antigravity, but qemu-user-aarch64 could not be installed" exit 1 } fi echo " Antigravity runtime prerequisites present." fi fi write_resolver_file() { local output_file="$1" mkdir -p "$(dirname "$output_file")" { found=0 for prop in net.dns1 net.dns2 net.dns3 net.dns4; do val=$(getprop "$prop" 2>/dev/null | tr -d '\r') if [ -n "$val" ] && echo "$val" | grep -qE '^([0-9]{1,3}\.){3}[0-9]{1,3}$|:'; then echo "nameserver $val" found=1 fi done if [ "$found" = "0" ]; then echo "nameserver 1.1.1.1" echo "nameserver 8.8.8.8" fi } > "$output_file" } # Replace "/etc/resolv.conf" with "/proc/self/fd/9\0" inside a binary # both are 16 bytes so file layout is preserved. patch_resolv_conf_binary() { local file="$1" off count=0 local offs offs=$(grep -abo "/etc/resolv.conf" "$file" | cut -d: -f1 || true) if [ -z "$offs" ]; then if grep -aq "/proc/self/fd/9" "$file"; then echo " already patched" return fi err "/etc/resolv.conf not found in binary" exit 1 fi for off in $offs; do printf '/proc/self/fd/9\0' | dd of="$file" bs=1 seek="$off" count=16 conv=notrunc status=none count=$((count + 1)) done echo " patched $count occurrence(s)" } install_claude() { section "Claude Code" # Claude Code 2.1.113+ moved to a native glibc binary. Android/Termux uses # Bionic libc, so those releases cannot run natively. 2.1.112 is the last # JS-based release known to work directly under Termux. CLAUDE_TERMUX_VERSION="${CLAUDE_TERMUX_VERSION:-2.1.112}" INSTALL_TAG="$CLAUDE_TERMUX_VERSION" step "claude 1/4" "Using Termux-compatible Claude Code release..." echo " Version: $CLAUDE_TERMUX_VERSION" if [ "${FORCE_INSTALL_CC_CODEX:-0}" != "1" ] \ && [ -f "$CLAUDE_VERSION_MARKER" ] \ && [ "$(cat "$CLAUDE_VERSION_MARKER")" = "$INSTALL_TAG" ] \ && [ -f "$CLAUDE_CLI" ]; then step "claude 2/4" "Already installed, skipping package install." else step "claude 2/4" "Installing JS-based Claude Code with npm..." rm -rf "$CLAUDE_PKG_DIR" rm -f "$USER_BIN/claude" mkdir -p "$CLAUDE_PKG_DIR" npm install --prefix "$CLAUDE_PKG_DIR" \ --omit=dev --no-audit --no-fund --loglevel=error \ "@anthropic-ai/claude-code@$CLAUDE_TERMUX_VERSION" [ -f "$CLAUDE_CLI" ] || { err "Claude cli.js not found after npm install"; exit 1; } echo "$INSTALL_TAG" > "$CLAUDE_VERSION_MARKER" fi step "claude 3/4" "Checking Claude entrypoint..." node "$CLAUDE_CLI" --version >/dev/null echo " Claude entrypoint OK" step "claude 4/4" "Writing Termux wrapper..." mkdir -p "$USER_BIN" cat > "$USER_BIN/claude" < "$USER_BIN/codex" <&2 exit 1 fi export SSL_CERT_FILE="\${SSL_CERT_FILE:-$PREFIX/etc/tls/cert.pem}" exec "\$CODEX_BIN" "\$@" 9<"\$CODEX_RESOLV_CONF" WRAPPER chmod +x "$USER_BIN/codex" echo "$INSTALL_TAG" > "$CODEX_VERSION_MARKER" } install_antigravity() { section "Antigravity CLI" # Google's official linux_arm64 Antigravity binary is glibc-linked and is # not natively executable on Android/Bionic. This uses the community Termux # port, which repackages the upstream Google release with Android/VA39 fixes. step "agy 1/4" "Resolving latest Termux-compatible Antigravity release..." if [ -n "${AGY_RELEASE_TAG:-}" ]; then TAG="$AGY_RELEASE_TAG" ARCHIVE_URL="https://github.com/$AGY_REPO/releases/download/$TAG/antigravity-termux-standalone.tar.gz" echo " Using tag override: $TAG" else META=$("${CURL[@]}" "https://api.github.com/repos/$AGY_REPO/releases/latest") TAG=$(echo "$META" | grep -o '"tag_name":[[:space:]]*"[^"]*"' | head -1 | sed 's/.*"\([^"]*\)"$/\1/') ARCHIVE_URL="https://github.com/$AGY_REPO/releases/latest/download/antigravity-termux-standalone.tar.gz" fi [ -n "$TAG" ] || { err "Could not resolve Antigravity Termux release tag"; exit 1; } echo " Tag: $TAG" INSTALL_TAG="$TAG" if [ "${FORCE_INSTALL_CC_CODEX:-0}" != "1" ] \ && [ -f "$AGY_VERSION_MARKER" ] \ && [ "$(cat "$AGY_VERSION_MARKER")" = "$INSTALL_TAG" ] \ && [ -x "$AGY_BIN" ] \ && [ -x "$AGY_VA39_BIN" ]; then step "agy 2/4" "Already up to date, skipping download." else step "agy 2/4" "Downloading Termux-patched Antigravity standalone build..." rm -rf "$AGY_DIR" rm -f "$USER_BIN/agy" "$USER_BIN/antigravity" mkdir -p "$AGY_DIR" "${CURL[@]}" "$ARCHIVE_URL" -o "$TMP_DIR/antigravity-termux.tgz" tar -xzf "$TMP_DIR/antigravity-termux.tgz" -C "$AGY_DIR" agy agy.va39 [ -f "$AGY_BIN" ] && [ -f "$AGY_VA39_BIN" ] || { err "Antigravity archive did not contain agy + agy.va39" exit 1 } chmod +x "$AGY_BIN" "$AGY_VA39_BIN" echo "$INSTALL_TAG" > "$AGY_VERSION_MARKER" fi step "agy 3/4" "Checking Antigravity entrypoint..." AGY_CHECK=$("$AGY_BIN" --version 2>&1) || { err "Antigravity installed, but failed to execute on this Termux device" printf '%s\n' "$AGY_CHECK" >&2 exit 1 } echo " ${AGY_CHECK:-Antigravity entrypoint OK}" step "agy 4/4" "Writing wrappers..." mkdir -p "$USER_BIN" cat > "$USER_BIN/agy" < "$USER_BIN/antigravity" <> "$rc" echo " $rc: added" } setup_shell_path() { section "Shell PATH" case ":${PATH:-}:" in *":$USER_BIN:"*) echo " current session PATH already includes $USER_BIN" ;; *) echo " current session PATH will include $USER_BIN after reloading your shell" ;; esac local sh_block fish_block sh_block="$PATH_MARKER_BEGIN case \":\$PATH:\" in *\":\$HOME/.local/bin:\"*) ;; *) export PATH=\"\$HOME/.local/bin:\$PATH\" ;; esac $PATH_MARKER_END" fish_block="$PATH_MARKER_BEGIN if not contains \"\$HOME/.local/bin\" \$PATH set -gx PATH \"\$HOME/.local/bin\" \$PATH end $PATH_MARKER_END" add_path_to_rc "$HOME/.bashrc" "$sh_block" add_path_to_rc "$HOME/.zshrc" "$sh_block" add_path_to_rc "$HOME/.config/fish/config.fish" "$fish_block" } uninstall_paths() { section "Uninstall $1" shift for p in "$@"; do [ -e "$p" ] || [ -L "$p" ] || continue rm -rf "$p" echo " removed $p" done } if [ "$action" = "uninstall" ]; then [ "$DO_CLAUDE" = "1" ] && uninstall_paths "Claude Code" \ "$USER_BIN/claude" "$CLAUDE_PKG_DIR" \ "$USER_LIB/glibc-claude" "$USER_LIB/musl-claude" "$USER_LIB/claude-code" \ "$HOME/.config/claude/resolv.conf" \ "$HOME/.config/claude/nsswitch.conf" \ "$HOME/.config/claude/hosts" [ "$DO_CODEX" = "1" ] && uninstall_paths "Codex CLI" "$USER_BIN/codex" "$CODEX_DIR" "$CODEX_RESOLV_CONF" [ "$DO_AGY" = "1" ] && uninstall_paths "Antigravity CLI" \ "$USER_BIN/agy" "$USER_BIN/antigravity" "$AGY_DIR" printf '\nDone. Auth/config dirs (~/.claude, ~/.codex, Antigravity settings) and rc-file PATH entries\n' echo "left intact." exit 0 fi [ "$DO_CLAUDE" = "1" ] && install_claude [ "$DO_CODEX" = "1" ] && install_codex [ "$DO_AGY" = "1" ] && install_antigravity setup_shell_path printf '\nDone.\n' case ":${PATH:-}:" in *":$USER_BIN:"*) ;; *) case "$(basename "${SHELL:-}")" in bash) echo " (Open a new shell or: source ~/.bashrc)" ;; zsh) echo " (Open a new shell or: source ~/.zshrc)" ;; fish) echo " (Open a new shell or: source ~/.config/fish/config.fish)" ;; *) echo " (Open a new shell to pick up the updated PATH)" ;; esac ;; esac if [ "$DO_CLAUDE" = "1" ]; then echo " Run: claude" fi if [ "$DO_CODEX" = "1" ]; then echo " Run: codex" fi if [ "$DO_AGY" = "1" ]; then echo " Run: agy (or: antigravity)" fi
